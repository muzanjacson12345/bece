  allow read: if signedIn() && (resource.data.uid == request.auth.uid || isOwner());
  allow write: if false;
}

  allow read: if signedIn() && (resource.data.uid == request.auth.uid || isStaff());

  allow read: if signedIn() && (resource.data.status == 'active' || isStaff());
  allow write: if false;
  match /questions/{questionId} { allow read, write: if false; }
  match /participants/{uid} {
    allow read: if signedIn() && (request.auth.uid == uid || isStaff());
    allow write: if false;
    match /attempts/{attemptId} { allow read, write: if false; }
  }
  match /leaderboard/{uid} { allow read: if signedIn(); allow write: if false; }
}
